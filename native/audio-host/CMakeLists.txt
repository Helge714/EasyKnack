cmake_minimum_required(VERSION 3.20)
project(EasyKnackAudioHost LANGUAGES CXX)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/build.properties" NATIVE_POLICY)
foreach(SETTING IN LISTS NATIVE_POLICY)
  if(SETTING MATCHES "^cpp.standard=(.*)$")
    set(CMAKE_CXX_STANDARD "${CMAKE_MATCH_1}")
  elseif(SETTING MATCHES "^windows.libraries=(.*)$")
    string(REPLACE " " ";" NATIVE_LIBRARIES "${CMAKE_MATCH_1}")
  elseif(SETTING MATCHES "^host.compile.flags=(.*)$")
    separate_arguments(NATIVE_FLAGS NATIVE_COMMAND "${CMAKE_MATCH_1}")
  endif()
endforeach()
if(MINGW)
  add_compile_options(${NATIVE_FLAGS})
  add_link_options(-static)
endif()
add_link_libraries(${NATIVE_LIBRARIES})
set(EASYKNACK_GENERATED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../target/native/generated" CACHE PATH "Maven-generated parameter contracts")
if(NOT EXISTS "${EASYKNACK_GENERATED_DIR}/ParameterContracts.hpp")
  message(FATAL_ERROR "Run Maven process-classes before configuring the native build")
endif()
include_directories("${EASYKNACK_GENERATED_DIR}")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "The initial local control transport targets Windows.")
endif()
file(GLOB_RECURSE NODE_HEADERS CONFIGURE_DEPENDS "src/nodes/*.hpp")
file(GLOB_RECURSE NODE_SOURCES CONFIGURE_DEPENDS "src/nodes/*.cpp")
set(NODE_GENERATOR "${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/generate-native-nodes.ps1")
set(NODE_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated/GeneratedNodeRegistry.hpp")
add_custom_target(generate-node-registry
  COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass -File "${NODE_GENERATOR}"
    -NodeDirectory "${CMAKE_CURRENT_SOURCE_DIR}/src/nodes" -OutputDirectory "${CMAKE_CURRENT_BINARY_DIR}/generated"
  BYPRODUCTS "${NODE_GENERATED}" DEPENDS ${NODE_HEADERS} "${NODE_GENERATOR}" VERBATIM)
add_executable(easyknack-audio-host src/main.cpp ${NODE_SOURCES})
add_dependencies(easyknack-audio-host generate-node-registry)
target_include_directories(easyknack-audio-host PRIVATE third_party src "${CMAKE_CURRENT_BINARY_DIR}/generated")
enable_testing()
add_executable(windows-audio-backend-test tests/WindowsAudioBackendTest.cpp ${NODE_SOURCES})
add_dependencies(windows-audio-backend-test generate-node-registry)
target_include_directories(windows-audio-backend-test PRIVATE third_party src "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_test(NAME windows-audio-backend COMMAND windows-audio-backend-test)
add_executable(asio-backend-test tests/AsioBackendTest.cpp ${NODE_SOURCES})
add_dependencies(asio-backend-test generate-node-registry)
target_include_directories(asio-backend-test PRIVATE third_party src "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_test(NAME asio-backend COMMAND asio-backend-test)
add_executable(audio-graph-test tests/AudioGraphTest.cpp ${NODE_SOURCES})
add_dependencies(audio-graph-test generate-node-registry)
target_include_directories(audio-graph-test PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_test(NAME audio-graph COMMAND audio-graph-test)
add_executable(shared-audio-transport-test tests/SharedAudioTransportTest.cpp)
if(MINGW)
  target_link_options(shared-audio-transport-test PRIVATE -municode)
endif()
add_test(NAME shared-audio-transport COMMAND shared-audio-transport-test)
add_executable(driver-bridge-test tests/DriverBridgeTest.cpp)
add_test(NAME driver-bridge COMMAND driver-bridge-test)
add_executable(driver-bridge-probe ../experimental/virtual-audio/BridgeProbe.cpp)
