cmake_minimum_required(VERSION 3.20)
project(EasyKnackAudioHost LANGUAGES CXX)
enable_language(C)
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
  set_source_files_properties(src/platform/MacPlatform.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
endif()
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/build.properties" NATIVE_POLICY)
if(WIN32)
  set(PLATFORM_KEY windows)
  set(POLICY_FLAGS "host.compile.flags")
elseif(APPLE)
  set(PLATFORM_KEY macos)
  set(POLICY_FLAGS "macos.host.compile.flags")
else()
  set(PLATFORM_KEY linux)
  set(POLICY_FLAGS "linux.host.compile.flags")
endif()
foreach(SETTING IN LISTS NATIVE_POLICY)
  if(SETTING MATCHES "^cpp.standard=(.*)$")
    set(CMAKE_CXX_STANDARD "${CMAKE_MATCH_1}")
  elseif(SETTING MATCHES "^${POLICY_FLAGS}=(.*)$")
    separate_arguments(NATIVE_FLAGS NATIVE_COMMAND "${CMAKE_MATCH_1}")
  elseif(SETTING MATCHES "^${PLATFORM_KEY}.libraries=(.*)$")
    separate_arguments(NATIVE_LIBRARIES NATIVE_COMMAND "${CMAKE_MATCH_1}")
  elseif(SETTING MATCHES "^${PLATFORM_KEY}.frameworks=(.*)$")
    separate_arguments(NATIVE_FRAMEWORKS NATIVE_COMMAND "${CMAKE_MATCH_1}")
  endif()
endforeach()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(EASYKNACK_GENERATED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../target/native/generated" CACHE PATH "Maven-generated parameter contracts")
option(EASYKNACK_PORTABLE_HOST "Test the portable backend on Windows instead of WASAPI/ASIO" OFF)
if(NOT EXISTS "${EASYKNACK_GENERATED_DIR}/ParameterContracts.hpp")
  message(FATAL_ERROR "Run Maven process-classes before configuring the native build")
endif()
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
file(GLOB LAME_SOURCES CONFIGURE_DEPENDS "third_party/lame/libmp3lame/*.c")
add_library(mp3lame SHARED ${LAME_SOURCES})
target_compile_definitions(mp3lame PRIVATE HAVE_CONFIG_H)
target_include_directories(mp3lame PRIVATE third_party/lame/easyknack third_party/lame/include third_party/lame/libmp3lame)
set_target_properties(mp3lame PROPERTIES SOVERSION 0 VERSION 0.0.0 C_STANDARD 99 C_STANDARD_REQUIRED ON)
if(UNIX)
  target_link_libraries(mp3lame PRIVATE m)
elseif(MINGW)
  target_link_options(mp3lame PRIVATE -static -Wl,--export-all-symbols)
endif()
file(GLOB_RECURSE NODE_HEADERS CONFIGURE_DEPENDS "src/nodes/*.hpp")
file(GLOB_RECURSE NODE_SOURCES CONFIGURE_DEPENDS "src/nodes/*.cpp")
if(APPLE)
  list(APPEND NODE_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/src/platform/MacPlatform.mm")
endif()
set(NODE_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated/GeneratedNodeRegistry.hpp")
add_custom_command(OUTPUT "${NODE_GENERATED}"
  COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/generate-native-nodes.py"
    --nodes "${CMAKE_CURRENT_SOURCE_DIR}/src/nodes" --output "${CMAKE_CURRENT_BINARY_DIR}/generated"
  DEPENDS ${NODE_HEADERS} "${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/generate-native-nodes.py" VERBATIM)
add_custom_target(generate-node-registry DEPENDS "${NODE_GENERATED}")
add_library(easyknack-platform INTERFACE)
target_include_directories(easyknack-platform INTERFACE third_party src "${EASYKNACK_GENERATED_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(easyknack-platform INTERFACE Threads::Threads ${NATIVE_LIBRARIES})
target_compile_options(easyknack-platform INTERFACE ${NATIVE_FLAGS})
if(WIN32)
  if(MINGW)
    target_link_options(easyknack-platform INTERFACE -static)
  endif()
elseif(APPLE)
  foreach(FRAMEWORK IN LISTS NATIVE_FRAMEWORKS)
    target_link_libraries(easyknack-platform INTERFACE "-framework ${FRAMEWORK}")
  endforeach()
endif()
function(host_target name source)
  add_executable(${name} ${source} ${NODE_SOURCES})
  add_dependencies(${name} generate-node-registry)
  target_link_libraries(${name} PRIVATE easyknack-platform)
endfunction()
host_target(easyknack-audio-host src/main.cpp)
if(EASYKNACK_PORTABLE_HOST)
  target_compile_definitions(easyknack-audio-host PRIVATE EASYKNACK_PORTABLE_HOST)
endif()
include(CTest)
if(BUILD_TESTING)
  add_test(NAME node-registry COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/test-native-generator.py")
  add_executable(midi-port-bank-test tests/MidiPortBankTest.cpp)
  target_link_libraries(midi-port-bank-test PRIVATE easyknack-platform)
  add_test(NAME midi-port-bank COMMAND midi-port-bank-test)
  add_executable(lame-recording-test tests/LameRecordingTest.cpp)
  target_link_libraries(lame-recording-test PRIVATE easyknack-platform)
  add_dependencies(lame-recording-test mp3lame)
  add_test(NAME lame-recording COMMAND lame-recording-test)
  if(UNIX AND NOT APPLE)
    set_tests_properties(lame-recording PROPERTIES ENVIRONMENT "LD_LIBRARY_PATH=$<TARGET_FILE_DIR:mp3lame>:$ENV{LD_LIBRARY_PATH}")
  elseif(APPLE)
    set_tests_properties(lame-recording PROPERTIES ENVIRONMENT "DYLD_LIBRARY_PATH=$<TARGET_FILE_DIR:mp3lame>:$ENV{DYLD_LIBRARY_PATH}")
  endif()
  host_target(audio-graph-test tests/AudioGraphTest.cpp)
  host_target(routed-audio-test tests/RoutedAudioTest.cpp)
  add_test(NAME routed-audio COMMAND routed-audio-test)
  host_target(linux-audio-adapter-test tests/LinuxAudioTest.cpp)
  add_test(NAME linux-audio-adapter COMMAND linux-audio-adapter-test)
  add_test(NAME audio-graph COMMAND audio-graph-test)
  host_target(portable-platform-test tests/PortablePlatformTest.cpp)
  add_library(platform-library-fixture SHARED tests/PlatformLibraryFixture.cpp)
  if(MINGW)
    target_link_options(platform-library-fixture PRIVATE -static)
  endif()
  add_test(NAME portable-platform COMMAND portable-platform-test "$<TARGET_FILE:platform-library-fixture>")
  add_test(NAME host-protocol COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/tests/HostProtocolTest.py" "$<TARGET_FILE:easyknack-audio-host>")
  add_executable(command-reader-test tests/CommandReaderTest.cpp)
  target_link_libraries(command-reader-test PRIVATE easyknack-platform)
  add_test(NAME command-reader COMMAND command-reader-test)
  if(APPLE)
    host_target(mac-midi-test tests/MacMidiTest.cpp)
    add_test(NAME mac-midi COMMAND mac-midi-test)
    set_tests_properties(mac-midi PROPERTIES SKIP_RETURN_CODE 77)
  endif()
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    host_target(linux-desktop-test tests/LinuxDesktopTest.cpp)
    add_test(NAME linux-desktop COMMAND linux-desktop-test)
    set_tests_properties(linux-desktop PROPERTIES SKIP_RETURN_CODE 77)
  endif()
  if(WIN32)
    host_target(windows-audio-backend-test tests/WindowsAudioBackendTest.cpp)
    host_target(asio-backend-test tests/AsioBackendTest.cpp)
    add_test(NAME windows-audio-backend COMMAND windows-audio-backend-test)
    add_test(NAME asio-backend COMMAND asio-backend-test)
    add_executable(shared-audio-transport-test tests/SharedAudioTransportTest.cpp)
    target_link_libraries(shared-audio-transport-test PRIVATE easyknack-platform)
    if(MINGW)
      target_link_options(shared-audio-transport-test PRIVATE -municode)
    endif()
    add_test(NAME shared-audio-transport COMMAND shared-audio-transport-test)
    add_executable(driver-bridge-test tests/DriverBridgeTest.cpp)
    add_test(NAME driver-bridge COMMAND driver-bridge-test)
    add_executable(driver-bridge-probe ../experimental/virtual-audio/BridgeProbe.cpp)
  endif()
  # Release host optimization must not turn assert-based acceptance checks off.
  get_property(NATIVE_TARGETS DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
  foreach(NATIVE_TARGET IN LISTS NATIVE_TARGETS)
    if(NATIVE_TARGET MATCHES "-test$")
      if(MSVC)
        target_compile_options(${NATIVE_TARGET} PRIVATE /UNDEBUG)
      else()
        target_compile_options(${NATIVE_TARGET} PRIVATE -UNDEBUG)
      endif()
    endif()
  endforeach()
endif()
